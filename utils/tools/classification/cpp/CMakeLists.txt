cmake_minimum_required(VERSION 3.16)
project(classification_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
add_executable(classification_benchmark main.cpp)
target_include_directories(classification_benchmark PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_options(classification_benchmark PRIVATE -O3 -Wall -Wextra)
if(TARGET_S100)
  set(NV12_ROW_ALIGN 32 CACHE STRING "NV12 row byte alignment: 32 for nash-e/nash-m, 64 for nash-p")
  target_compile_definitions(classification_benchmark PRIVATE TARGET_S100=1 NV12_ROW_ALIGN=${NV12_ROW_ALIGN})
  target_link_directories(classification_benchmark PRIVATE /usr/hobot/lib)
  target_link_libraries(classification_benchmark PRIVATE dnn hbucp ${OpenCV_LIBS} Threads::Threads)
else()
  target_link_libraries(classification_benchmark PRIVATE dnn ${OpenCV_LIBS} Threads::Threads)
endif()
