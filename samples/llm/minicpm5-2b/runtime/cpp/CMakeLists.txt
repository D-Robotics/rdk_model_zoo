cmake_minimum_required(VERSION 3.16)
project(minicpm5_sample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OELLM_RUNTIME_ROOT "" CACHE PATH "Path to the supplied oellm_runtime directory")
if(NOT EXISTS "${OELLM_RUNTIME_ROOT}/include/oellm_runtime_basic/oellm_runtime.h")
  message(FATAL_ERROR "Set OELLM_RUNTIME_ROOT to the SDK's oellm_runtime directory")
endif()
file(READ "/sys/class/boardinfo/soc_name" SOC_NAME_RAW)
string(STRIP "${SOC_NAME_RAW}" SOC_NAME)
string(TOUPPER "${SOC_NAME}" SOC_NAME)
if(NOT SOC_NAME STREQUAL "S600")
  message(FATAL_ERROR "This sample artifact is validated for S600 only")
endif()
find_package(gflags REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(main src/main.cc src/minicpm5.cc)
target_compile_definitions(main PRIVATE SOC_S600)
target_compile_options(main PRIVATE -Wall -Wextra -Werror)
target_include_directories(main PRIVATE inc "${OELLM_RUNTIME_ROOT}/include")
target_link_libraries(main PRIVATE gflags nlohmann_json::nlohmann_json
  "${OELLM_RUNTIME_ROOT}/lib/liboellm_runtime.so"
  "${OELLM_RUNTIME_ROOT}/lib/libvla_sdk.so")
target_link_options(main PRIVATE "-Wl,-rpath-link,${OELLM_RUNTIME_ROOT}/lib")
set_target_properties(main PROPERTIES BUILD_RPATH "${OELLM_RUNTIME_ROOT}/lib")
