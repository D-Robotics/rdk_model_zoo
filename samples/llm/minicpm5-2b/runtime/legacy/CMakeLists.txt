cmake_minimum_required(VERSION 3.16)
project(minicpm5_legacy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OELLM_RUNTIME_ROOT "" CACHE PATH "SDK 1.0.0 oellm_runtime directory")
if(NOT EXISTS "${OELLM_RUNTIME_ROOT}/include/xlm.h" OR NOT EXISTS "${OELLM_RUNTIME_ROOT}/lib/libxlm.so")
  message(FATAL_ERROR "Set OELLM_RUNTIME_ROOT to SDK 1.0.0 oellm_runtime")
endif()
add_executable(main src/main.cc src/minicpm5.cc)
target_include_directories(main PRIVATE inc "${OELLM_RUNTIME_ROOT}/include")
target_link_libraries(main PRIVATE "${OELLM_RUNTIME_ROOT}/lib/libxlm.so")
target_link_options(main PRIVATE -Wl,--allow-shlib-undefined)
target_compile_options(main PRIVATE -Wall -Wextra -Werror)
