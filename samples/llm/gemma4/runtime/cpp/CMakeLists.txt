cmake_minimum_required(VERSION 3.16)
project(gemma4_text_sample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OELLM_RUNTIME_ROOT "" CACHE PATH "SDK 2.0 oellm_runtime_s600 directory")
if(NOT EXISTS "${OELLM_RUNTIME_ROOT}/include/oellm_runtime_basic/oellm_runtime.h")
  message(FATAL_ERROR "Set OELLM_RUNTIME_ROOT to SDK 2.0 oellm_runtime_s600")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Build for aarch64 using the S600 SDK or on an S600 board")
endif()
find_library(OELLM_LIBRARY oellm_runtime PATHS "${OELLM_RUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT OELLM_LIBRARY)
  message(FATAL_ERROR "The SDK deployment package is missing liboellm_runtime.so")
endif()
add_executable(main src/main.cc src/gemma4.cc)
target_include_directories(main PRIVATE inc "${OELLM_RUNTIME_ROOT}/include" "${OELLM_RUNTIME_ROOT}/3rdparty")
target_compile_options(main PRIVATE -Wall -Wextra -Werror)
target_link_libraries(main PRIVATE "${OELLM_LIBRARY}")
# Same transitive shared-library handling as the SDK's samples/basic target.
target_link_options(main PRIVATE "-Wl,-rpath-link,${OELLM_RUNTIME_ROOT}/lib" "-Wl,--unresolved-symbols=ignore-in-shared-libs")
set_target_properties(main PROPERTIES BUILD_RPATH "${OELLM_RUNTIME_ROOT}/lib")
