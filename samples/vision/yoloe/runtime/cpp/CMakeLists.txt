cmake_minimum_required(VERSION 3.16)
project(yoloe_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(YOLOE_BUILD_TESTS "Build host tests alongside the stage library" OFF)
option(YOLOE_SANITIZERS "Instrument the stage library and tests" OFF)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
get_filename_component(YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../ultralytics_yolo/runtime/cpp" ABSOLUTE)
get_filename_component(RDK_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../utils/c_utils" ABSOLUTE)
# Host-testable model core: geometry, NV12 preparation, E11/E26 decodes, mask
# restoration, preflight and the YOLOE stages. No board SDK symbols; the DNN
# adapter region of src/detect.cpp stays compiled out without YOLOE_HAS_SDK.
add_library(yoloe_core STATIC src/detect.cpp "${RDK_SHARED_CPP}/platform_identity.cc")
target_include_directories(yoloe_core PUBLIC inc "${YOLO_CPP}/inc" "${RDK_SHARED_CPP}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yoloe_core PUBLIC ${OpenCV_LIBS})
target_compile_options(yoloe_core PRIVATE -Wall -Wextra -Werror)
if(YOLOE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(yoloe_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(yoloe_core PUBLIC -fsanitize=address,undefined)
endif()
option(YOLOE_BUILD_CLI "Build the native executable (requires board SDK)" OFF)
option(YOLOE_BUILD_SDK "Build adapter against an installed board DNN SDK" OFF)
if(YOLOE_BUILD_CLI)
  set(YOLOE_BUILD_SDK ON)
endif()
if(YOLOE_BUILD_SDK)
  find_path(YOLOE_DNN_INCLUDE_DIR dnn/hb_dnn.h PATHS /usr/include /usr/include/hobot /usr/hobot/include)
  find_library(YOLOE_DNN_LIBRARY dnn PATHS /usr/lib /usr/hobot/lib)
  if(NOT YOLOE_DNN_INCLUDE_DIR OR NOT YOLOE_DNN_LIBRARY)
    message(FATAL_ERROR "YOLOE SDK adapter requires board dnn headers/library; set YOLOE_DNN_INCLUDE_DIR and YOLOE_DNN_LIBRARY")
  endif()
  # Same model sources with the DNN adapter enabled, plus the ultralytics_yolo
  # shared backend implementation. Self-contained: never link yoloe_core here,
  # the model translation unit is compiled once per library.
  add_library(yoloe_sdk STATIC src/detect.cpp "${YOLO_CPP}/src/backend.cpp" "${RDK_SHARED_CPP}/platform_identity.cc")
  target_compile_definitions(yoloe_sdk PRIVATE YOLOE_HAS_SDK=1)
  target_include_directories(yoloe_sdk PUBLIC inc "${YOLO_CPP}/inc" "${RDK_SHARED_CPP}" ${OpenCV_INCLUDE_DIRS})
  target_include_directories(yoloe_sdk PRIVATE "${YOLOE_DNN_INCLUDE_DIR}")
  target_compile_options(yoloe_sdk PRIVATE -Wall -Wextra -Werror)
  if(YOLOE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(yoloe_sdk PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  target_link_libraries(yoloe_sdk PUBLIC ${OpenCV_LIBS} "${YOLOE_DNN_LIBRARY}")
  if(EXISTS "${YOLOE_DNN_INCLUDE_DIR}/hb_ucp_sys.h")
    find_library(YOLOE_UCP_LIBRARY hbucp PATHS /usr/lib /usr/hobot/lib)
    if(NOT YOLOE_UCP_LIBRARY)
      message(FATAL_ERROR "UCP headers require matching hbucp library")
    endif()
    target_link_libraries(yoloe_sdk PUBLIC "${YOLOE_UCP_LIBRARY}")
  endif()
endif()
if(YOLOE_BUILD_CLI)
  add_executable(yoloe_demo src/main.cpp src/cli.cpp)
  target_link_libraries(yoloe_demo PRIVATE yoloe_sdk)
  target_compile_options(yoloe_demo PRIVATE -Wall -Wextra -Werror)
  if(YOLOE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(yoloe_demo PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
endif()
if(YOLOE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
