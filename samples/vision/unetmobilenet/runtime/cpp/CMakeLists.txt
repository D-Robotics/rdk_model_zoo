cmake_minimum_required(VERSION 3.16)
project(unetmobilenet_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RDK_TARGET "" CACHE STRING "Explicit board target: s100 or s600")
if(RDK_TARGET STREQUAL "auto")
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "RDK_TARGET=auto would read the build host SoC identity while cross-compiling; pass an explicit -DRDK_TARGET=s100|s600")
  endif()
  file(READ "/sys/class/boardinfo/soc_name" SOC_IDENTITY)
  string(STRIP "${SOC_IDENTITY}" SOC_IDENTITY)
  string(TOUPPER "${SOC_IDENTITY}" SOC_IDENTITY)
  if(NOT SOC_IDENTITY MATCHES "^(S100|S600)$")
    message(FATAL_ERROR "Unsupported RDK board target '${SOC_IDENTITY}': UnetMobileNet publishes S100 and S600 artifacts only")
  endif()
  set(RDK_TARGET_NORMALIZED "${SOC_IDENTITY}")
else()
  string(TOUPPER "${RDK_TARGET}" RDK_TARGET_NORMALIZED)
endif()
if(NOT RDK_TARGET_NORMALIZED MATCHES "^(S100|S600)$")
  message(FATAL_ERROR "Unsupported RDK board target '${RDK_TARGET}': UnetMobileNet publishes S100 and S600 artifacts only")
endif()
string(TOLOWER "${RDK_TARGET_NORMALIZED}" UNETMOBILENET_TARGET)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_executable(unetmobilenet_cpp src/main.cpp src/segment.cpp src/cli.cpp)
target_compile_definitions(unetmobilenet_cpp PRIVATE UNETMOBILENET_TARGET_NAME="${UNETMOBILENET_TARGET}")
target_compile_options(unetmobilenet_cpp PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(unetmobilenet_cpp PRIVATE inc ${OpenCV_INCLUDE_DIRS}
                           /usr/hobot/include /usr/include/hobot /usr/include/hobot/dnn)
target_link_directories(unetmobilenet_cpp PRIVATE /usr/hobot/lib)
target_link_libraries(unetmobilenet_cpp PRIVATE dnn hbucp ${OpenCV_LIBS})
