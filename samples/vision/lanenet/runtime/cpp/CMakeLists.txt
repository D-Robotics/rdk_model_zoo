cmake_minimum_required(VERSION 3.16)
project(lanenet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RDK_TARGET "auto" CACHE STRING "RDK board target: s100 (LaneNet publishes an S100 artifact and binding only); auto reads the on-board SoC identity")
if(RDK_TARGET STREQUAL "auto")
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "RDK_TARGET=auto would read the build host SoC identity while cross-compiling; pass an explicit target (-DRDK_TARGET=s100)")
  endif()
  file(READ "/sys/class/boardinfo/soc_name" RDK_SOC_NAME)
  string(STRIP "${RDK_SOC_NAME}" RDK_SOC_NAME)
  string(TOUPPER "${RDK_SOC_NAME}" RDK_SOC_NAME)
  set(RDK_TARGET "${RDK_SOC_NAME}")
endif()
string(TOUPPER "${RDK_TARGET}" RDK_TARGET_NORMALIZED)
if(NOT RDK_TARGET_NORMALIZED STREQUAL "S100")
  message(FATAL_ERROR "Unsupported RDK board target '${RDK_TARGET}': LaneNet publishes an S100 artifact and binding only")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
find_path(DNN_INCLUDE_DIR hobot/dnn/hb_dnn.h HINTS /usr/include /usr/hobot/include)
find_path(UCP_INCLUDE_DIR hobot/hb_ucp.h HINTS /usr/include /usr/hobot/include)
find_library(DNN_LIBRARY dnn HINTS /usr/hobot/lib /usr/lib)
find_library(UCP_LIBRARY hbucp HINTS /usr/hobot/lib /usr/lib)
if(NOT DNN_INCLUDE_DIR OR NOT UCP_INCLUDE_DIR OR NOT DNN_LIBRARY OR NOT UCP_LIBRARY)
  message(FATAL_ERROR "Prepare matching S100 DNN/UCP headers and libraries separately")
endif()
add_executable(lanenet src/main.cpp src/segment.cpp src/cli.cpp)
target_include_directories(lanenet PRIVATE inc ${DNN_INCLUDE_DIR} ${UCP_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
target_compile_options(lanenet PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lanenet PRIVATE ${DNN_LIBRARY} ${UCP_LIBRARY} ${OpenCV_LIBS} Threads::Threads ${CMAKE_DL_LIBS})
