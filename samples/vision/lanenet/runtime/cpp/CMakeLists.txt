cmake_minimum_required(VERSION 3.16)
project(lanenet LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LANENET_TARGET "s100" CACHE STRING "Only the published S100 LaneNet contract is supported")
if(NOT LANENET_TARGET STREQUAL "s100")
  message(FATAL_ERROR "LaneNet currently has only an S100 artifact and binding")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
find_path(DNN_INCLUDE_DIR hobot/dnn/hb_dnn.h HINTS /usr/include /usr/hobot/include)
find_path(UCP_INCLUDE_DIR hobot/hb_ucp.h HINTS /usr/include /usr/hobot/include)
find_library(DNN_LIBRARY dnn HINTS /usr/hobot/lib /usr/lib)
find_library(UCP_LIBRARY hbucp HINTS /usr/hobot/lib /usr/lib)
if(NOT DNN_INCLUDE_DIR OR NOT UCP_INCLUDE_DIR OR NOT DNN_LIBRARY OR NOT UCP_LIBRARY)
  message(FATAL_ERROR "Prepare matching S100 DNN/UCP headers and libraries separately")
endif()
add_library(lanenet_core STATIC src/tensor_contract.cpp src/lanenet.cpp src/visualization.cpp)
target_include_directories(lanenet_core PUBLIC inc ${OpenCV_INCLUDE_DIRS})
target_link_libraries(lanenet_core PUBLIC ${OpenCV_LIBS})
target_compile_options(lanenet_core PRIVATE -Wall -Wextra -Werror)
add_executable(lanenet src/main.cpp src/model_runner.cpp src/cli_io.cpp)
target_include_directories(lanenet PRIVATE ${DNN_INCLUDE_DIR} ${UCP_INCLUDE_DIR})
target_compile_options(lanenet PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lanenet PRIVATE lanenet_core ${DNN_LIBRARY} ${UCP_LIBRARY} Threads::Threads ${CMAKE_DL_LIBS})
