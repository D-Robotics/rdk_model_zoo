cmake_minimum_required(VERSION 3.16)
project(yolo26_depth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
find_path(DNN_INCLUDE_DIR dnn/hb_dnn.h HINTS /usr/include)
find_library(DNN_LIBRARY dnn HINTS /usr/lib)
if(NOT DNN_INCLUDE_DIR OR NOT DNN_LIBRARY)
  message(FATAL_ERROR "Install the matching RDK X5 SDK separately; dnn headers/library are required")
endif()
add_library(yolo26_depth_core STATIC src/yolo26_depth.cpp src/image_io.cpp src/tensor_contract.cpp)
target_include_directories(yolo26_depth_core PUBLIC inc ${OpenCV_INCLUDE_DIRS})
target_link_libraries(yolo26_depth_core PUBLIC ${OpenCV_LIBS})
target_compile_options(yolo26_depth_core PRIVATE -Wall -Wextra -Werror)
add_executable(yolo26_depth src/main.cpp src/model_runner.cpp src/cli_io.cpp)
target_include_directories(yolo26_depth PRIVATE ${DNN_INCLUDE_DIR})
target_compile_options(yolo26_depth PRIVATE -Wall -Wextra -Werror)
target_link_libraries(yolo26_depth PRIVATE yolo26_depth_core ${DNN_LIBRARY} Threads::Threads rt dl)
