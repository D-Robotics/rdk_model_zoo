cmake_minimum_required(VERSION 3.16)
project(yolo26_depth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RDK_TARGET "" CACHE STRING "Explicit board target: x5")
if(RDK_TARGET STREQUAL "auto")
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "RDK_TARGET=auto would read the build host SoC identity while cross-compiling; pass an explicit -DRDK_TARGET=x5")
  endif()
  file(READ "/sys/class/boardinfo/soc_name" SOC_IDENTITY)
  string(STRIP "${SOC_IDENTITY}" SOC_IDENTITY)
  string(TOUPPER "${SOC_IDENTITY}" SOC_IDENTITY)
  if(NOT SOC_IDENTITY MATCHES "^(X5)$")
    message(FATAL_ERROR "Unsupported RDK board target '${SOC_IDENTITY}': yolo26_depth publishes X5 artifacts only")
  endif()
  set(RDK_TARGET_NORMALIZED "${SOC_IDENTITY}")
else()
  string(TOUPPER "${RDK_TARGET}" RDK_TARGET_NORMALIZED)
endif()
if(NOT RDK_TARGET_NORMALIZED MATCHES "^(X5)$")
  message(FATAL_ERROR "Unsupported RDK board target '${RDK_TARGET}': yolo26_depth publishes X5 artifacts only")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
find_path(DNN_INCLUDE_DIR dnn/hb_dnn.h HINTS /usr/include)
find_library(DNN_LIBRARY dnn HINTS /usr/lib)
if(NOT DNN_INCLUDE_DIR OR NOT DNN_LIBRARY)
  message(FATAL_ERROR "Install the matching RDK X5 SDK separately; dnn headers/library are required")
endif()
add_executable(yolo26_depth src/main.cpp src/depth.cpp src/cli.cpp)
target_include_directories(yolo26_depth PRIVATE inc ${DNN_INCLUDE_DIR} ${OpenCV_INCLUDE_DIRS})
target_compile_options(yolo26_depth PRIVATE -Wall -Wextra -Werror)
target_link_libraries(yolo26_depth PRIVATE ${DNN_LIBRARY} ${OpenCV_LIBS} Threads::Threads rt dl)
