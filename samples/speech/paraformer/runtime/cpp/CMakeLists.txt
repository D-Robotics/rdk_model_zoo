cmake_minimum_required(VERSION 3.18)
project(paraformer_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PARAFORMER_BUILD_TESTS "Build host numerical contract tests" OFF)
option(PARAFORMER_SANITIZERS "Enable address/undefined sanitizers" OFF)
add_library(paraformer_contract STATIC src/contract.cc src/pipeline.cc)
target_include_directories(paraformer_contract PUBLIC inc)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(paraformer_contract PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(PARAFORMER_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(paraformer_contract PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(paraformer_contract PUBLIC -fsanitize=address,undefined)
endif()
if(PARAFORMER_BUILD_TESTS)
  enable_testing()
  add_executable(test_contract tests/test_contract.cc)
  target_link_libraries(test_contract PRIVATE paraformer_contract)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_contract PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME paraformer_contract COMMAND test_contract)
  add_executable(test_pipeline tests/test_pipeline.cc)
  target_link_libraries(test_pipeline PRIVATE paraformer_contract)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_pipeline PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME paraformer_pipeline COMMAND test_pipeline)
endif()

get_filename_component(PARAFORMER_YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vision/ultralytics_yolo/runtime/cpp" ABSOLUTE)
option(PARAFORMER_BUILD_SDK "Build S100 vendor SDK adapter" OFF)
if(PARAFORMER_BUILD_SDK)
  find_path(PARAFORMER_DNN_INCLUDE dnn/hb_dnn.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(PARAFORMER_UCP_INCLUDE hb_ucp.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(PARAFORMER_UCP_SYS_INCLUDE hb_ucp_sys.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_library(PARAFORMER_DNN_LIBRARY dnn PATHS /usr/hobot/lib REQUIRED)
  find_library(PARAFORMER_UCP_LIBRARY hbucp PATHS /usr/hobot/lib REQUIRED)
  add_library(paraformer_sdk STATIC src/sdk_runner.cc "${PARAFORMER_YOLO_CPP}/common/dnn_io.cc" "${PARAFORMER_YOLO_CPP}/common/nv12_geometry.cc")
  target_include_directories(paraformer_sdk PUBLIC inc PRIVATE "${PARAFORMER_YOLO_CPP}" ${PARAFORMER_DNN_INCLUDE} ${PARAFORMER_UCP_INCLUDE} ${PARAFORMER_UCP_SYS_INCLUDE})
  target_link_libraries(paraformer_sdk PUBLIC paraformer_contract ${PARAFORMER_DNN_LIBRARY} ${PARAFORMER_UCP_LIBRARY})
endif()
if(PARAFORMER_BUILD_TESTS)
  add_executable(test_sdk tests/test_sdk.cc src/sdk_runner.cc "${PARAFORMER_YOLO_CPP}/common/dnn_io.cc" "${PARAFORMER_YOLO_CPP}/common/nv12_geometry.cc")
  target_include_directories(test_sdk BEFORE PRIVATE "${PARAFORMER_YOLO_CPP}/test/fake_dnn_io/ucp")
  target_include_directories(test_sdk PRIVATE inc "${PARAFORMER_YOLO_CPP}")
  target_compile_definitions(test_sdk PRIVATE TEST_TENSOR_AWARE_INFER=1)
  target_link_libraries(test_sdk PRIVATE paraformer_contract)
  target_compile_options(test_sdk PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME paraformer_sdk_fixture COMMAND test_sdk)
endif()

get_filename_component(PARAFORMER_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../utils/c_utils" ABSOLUTE)

  target_link_libraries(test_preflight PRIVATE paraformer_contract)
  target_compile_options(test_preflight PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME paraformer_preflight COMMAND test_preflight "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../docs/releases/unified-migration/evidence/2026-09-28-b10-paraformer-pipeline/published-tokens.json")
