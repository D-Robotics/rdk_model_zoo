cmake_minimum_required(VERSION 3.18)
project(paraformer_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(PARAFORMER_YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vision/ultralytics_yolo/runtime/cpp" ABSOLUTE)
get_filename_component(PARAFORMER_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../utils/c_utils" ABSOLUTE)
option(PARAFORMER_BUILD_TESTS "Build host numerical contract tests" OFF)
option(PARAFORMER_SANITIZERS "Enable address/undefined sanitizers" OFF)
# Single model library: stage callables, CIF, decode contract, preflight and
# the UCP adapter behind the explicit PARAFORMER_ENABLE_UCP gate.
add_library(paraformer_pipeline STATIC src/pipeline.cpp src/cif.cpp "${PARAFORMER_SHARED_CPP}/platform_identity.cc")
target_include_directories(paraformer_pipeline PUBLIC inc "${PARAFORMER_SHARED_CPP}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(paraformer_pipeline PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
if(PARAFORMER_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(paraformer_pipeline PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(paraformer_pipeline PUBLIC -fsanitize=address,undefined)
endif()
if(PARAFORMER_BUILD_TESTS)
  enable_testing()
  add_executable(test_contract tests/test_contract.cc)
  target_link_libraries(test_contract PRIVATE paraformer_pipeline)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_contract PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME paraformer_contract COMMAND test_contract)
  add_executable(test_pipeline tests/test_pipeline.cc)
  target_link_libraries(test_pipeline PRIVATE paraformer_pipeline)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_pipeline PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  endif()
  add_test(NAME paraformer_pipeline COMMAND test_pipeline)
endif()

option(PARAFORMER_BUILD_SDK "Build S100 vendor SDK adapter" OFF)
if(PARAFORMER_BUILD_SDK)
  find_path(PARAFORMER_DNN_INCLUDE dnn/hb_dnn.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(PARAFORMER_UCP_INCLUDE hb_ucp.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(PARAFORMER_UCP_SYS_INCLUDE hb_ucp_sys.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_library(PARAFORMER_DNN_LIBRARY dnn PATHS /usr/hobot/lib REQUIRED)
  find_library(PARAFORMER_UCP_LIBRARY hbucp PATHS /usr/hobot/lib REQUIRED)
  target_sources(paraformer_pipeline PRIVATE "${PARAFORMER_YOLO_CPP}/src/backend.cpp")
  target_include_directories(paraformer_pipeline PRIVATE "${PARAFORMER_YOLO_CPP}" "${PARAFORMER_YOLO_CPP}/inc" ${PARAFORMER_DNN_INCLUDE} ${PARAFORMER_UCP_INCLUDE} ${PARAFORMER_UCP_SYS_INCLUDE})
  target_link_libraries(paraformer_pipeline PRIVATE ${PARAFORMER_DNN_LIBRARY} ${PARAFORMER_UCP_LIBRARY})
  # Explicit compile-time selection only: visible vendor headers alone must
  # never enable SDK bindings in SDK-free builds.
  target_compile_definitions(paraformer_pipeline PRIVATE PARAFORMER_ENABLE_UCP=1)
endif()
if(PARAFORMER_BUILD_TESTS)
  # The fake-SDK suite compiles the model library itself with the declaration-
  # only fakes deliberately selected, exactly as the board build selects the
  # real UCP stack.
  add_executable(test_sdk tests/test_sdk.cc src/pipeline.cpp src/cif.cpp "${PARAFORMER_SHARED_CPP}/platform_identity.cc" "${PARAFORMER_YOLO_CPP}/src/backend.cpp")
  target_include_directories(test_sdk BEFORE PRIVATE "${PARAFORMER_YOLO_CPP}/test/fake_dnn_io/ucp")
  target_include_directories(test_sdk PRIVATE inc "${PARAFORMER_SHARED_CPP}" "${PARAFORMER_YOLO_CPP}" "${PARAFORMER_YOLO_CPP}/inc")
  target_compile_definitions(test_sdk PRIVATE TEST_TENSOR_AWARE_INFER=1 PARAFORMER_ENABLE_UCP=1)
  target_compile_options(test_sdk PRIVATE -Wall -Wextra -Werror -UNDEBUG -ffp-contract=off)
  add_test(NAME paraformer_sdk_fixture COMMAND test_sdk)
endif()

if(PARAFORMER_BUILD_TESTS)
  # Compiles the model sources directly because this suite injects its own
  # mutable local-identity double instead of the real platform probe.
  add_executable(test_preflight tests/test_preflight.cc src/pipeline.cpp src/cif.cpp)
  target_include_directories(test_preflight PRIVATE inc "${PARAFORMER_SHARED_CPP}")
  target_compile_options(test_preflight PRIVATE -Wall -Wextra -Werror -UNDEBUG -ffp-contract=off)
  if(PARAFORMER_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_preflight PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_preflight PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME paraformer_preflight COMMAND test_preflight "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/published-tokens.json")
endif()

option(PARAFORMER_BUILD_IO "Build prepared-manifest/NPY reader (requires nlohmann JSON)" OFF)
if(PARAFORMER_BUILD_IO)
  find_path(PARAFORMER_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  if(PARAFORMER_BUILD_TESTS)
    add_executable(feature_probe tests/feature_probe.cc src/cli.cpp)
    target_include_directories(feature_probe PRIVATE "${PARAFORMER_SHARED_CPP}" ${PARAFORMER_JSON_INCLUDE})
    target_link_libraries(feature_probe PRIVATE paraformer_pipeline)
    add_test(NAME paraformer_prepared_features COMMAND feature_probe "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fixtures/features/prepared-manifest.json")
  endif()
endif()

option(PARAFORMER_BUILD_CLI "Build complete S100 native executable" OFF)
if(PARAFORMER_BUILD_CLI)
  if(NOT PARAFORMER_BUILD_SDK OR NOT PARAFORMER_BUILD_IO)
    message(FATAL_ERROR "PARAFORMER_BUILD_CLI requires PARAFORMER_BUILD_SDK=ON and PARAFORMER_BUILD_IO=ON")
  endif()
  add_executable(paraformer_demo src/main.cpp src/cli.cpp)
  target_include_directories(paraformer_demo PRIVATE ${PARAFORMER_JSON_INCLUDE})
  target_link_libraries(paraformer_demo PRIVATE paraformer_pipeline)
endif()
if(PARAFORMER_BUILD_TESTS AND PARAFORMER_BUILD_IO)
  # Host CLI fixture: the same main/model/CLI sources over a clearly marked
  # transport/identity double instead of any SDK or platform probe.
  add_executable(paraformer_cli_fixture src/main.cpp src/cli.cpp src/pipeline.cpp src/cif.cpp tests/cli_fixture_runner.cc)
  target_compile_definitions(paraformer_cli_fixture PRIVATE PARAFORMER_HOST_FIXTURE=1)
  target_include_directories(paraformer_cli_fixture PRIVATE inc "${PARAFORMER_SHARED_CPP}" ${PARAFORMER_JSON_INCLUDE})
  target_compile_options(paraformer_cli_fixture PRIVATE -Wall -Wextra -Werror -UNDEBUG -ffp-contract=off)
  add_test(NAME paraformer_cli_fixture_help COMMAND paraformer_cli_fixture --help)
endif()
