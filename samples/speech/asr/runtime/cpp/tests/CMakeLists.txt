cmake_minimum_required(VERSION 3.18)
project(asr_contract_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASR_SANITIZERS "Enable ASan/UBSan" ON)
enable_testing()
add_executable(test_contract test_contract.cc)
target_include_directories(test_contract PRIVATE ../inc)
target_compile_options(test_contract PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(test_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test_contract PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME asr_contract COMMAND test_contract)

option(ASR_AUDIO_TESTS "Build tests using actual libsndfile/libsamplerate" OFF)
if(ASR_AUDIO_TESTS)
  find_path(ASR_SNDFILE_INCLUDE sndfile.h REQUIRED)
  find_path(ASR_SAMPLERATE_INCLUDE samplerate.h REQUIRED)
  find_library(ASR_SNDFILE_LIBRARY sndfile REQUIRED)
  find_library(ASR_SAMPLERATE_LIBRARY samplerate REQUIRED)
  add_executable(test_audio test_audio.cc ../src/audio_io.cc ../src/frontend.cc)
  target_include_directories(test_audio PRIVATE ../inc ${ASR_SNDFILE_INCLUDE} ${ASR_SAMPLERATE_INCLUDE})
  target_link_libraries(test_audio PRIVATE ${ASR_SNDFILE_LIBRARY} ${ASR_SAMPLERATE_LIBRARY})
  target_compile_options(test_audio PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_audio PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_audio PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME asr_audio COMMAND test_audio)
endif()
if(ASR_AUDIO_TESTS)
  add_executable(test_task test_task.cc ../src/frontend.cc)
  target_include_directories(test_task PRIVATE ../inc ${ASR_SAMPLERATE_INCLUDE})

    target_link_options(test_task PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME asr_task COMMAND test_task)
