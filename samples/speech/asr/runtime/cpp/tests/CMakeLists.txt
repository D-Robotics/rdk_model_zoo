cmake_minimum_required(VERSION 3.18)
project(asr_contract_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASR_SANITIZERS "Enable ASan/UBSan" ON)
enable_testing()
get_filename_component(ASR_YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../vision/ultralytics_yolo/runtime/cpp" ABSOLUTE)
get_filename_component(ASR_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../utils/c_utils" ABSOLUTE)
# src/asr.cpp owns preflight, the model task and the UCP binding, so every
# binary that links it also links the audio frontend (src/frontend.cpp) and
# therefore libsamplerate.
find_path(ASR_SAMPLERATE_INCLUDE samplerate.h REQUIRED)
find_library(ASR_SAMPLERATE_LIBRARY samplerate REQUIRED)

add_executable(test_contract test_contract.cc)
target_include_directories(test_contract PRIVATE ../inc "${ASR_SHARED_CPP}")
target_compile_options(test_contract PRIVATE -Wall -Wextra -Werror -UNDEBUG)
if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(test_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(test_contract PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME asr_contract COMMAND test_contract)

option(ASR_AUDIO_TESTS "Build tests using actual libsndfile/libsamplerate" OFF)
if(ASR_AUDIO_TESTS)
  find_path(ASR_SNDFILE_INCLUDE sndfile.h REQUIRED)
  find_path(ASR_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  find_library(ASR_SNDFILE_LIBRARY sndfile REQUIRED)
  add_executable(test_audio test_audio.cc ../src/cli.cpp ../src/asr.cpp
    ../src/frontend.cpp "${ASR_SHARED_CPP}/platform_identity.cc")
  target_include_directories(test_audio PRIVATE ../inc "${ASR_SHARED_CPP}" ${ASR_SNDFILE_INCLUDE} ${ASR_JSON_INCLUDE} ${ASR_SAMPLERATE_INCLUDE})
  target_link_libraries(test_audio PRIVATE ${ASR_SNDFILE_LIBRARY} ${ASR_SAMPLERATE_LIBRARY})
  target_compile_options(test_audio PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_audio PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_audio PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME asr_audio COMMAND test_audio)
  add_executable(test_task test_task.cc ../src/asr.cpp ../src/frontend.cpp
    "${ASR_SHARED_CPP}/platform_identity.cc")
  target_include_directories(test_task PRIVATE ../inc "${ASR_SHARED_CPP}" ${ASR_SAMPLERATE_INCLUDE})
  target_link_libraries(test_task PRIVATE ${ASR_SAMPLERATE_LIBRARY})
  target_compile_options(test_task PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(test_task PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(test_task PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME asr_task COMMAND test_task)
endif()

add_executable(test_sdk test_sdk.cc ../src/asr.cpp ../src/frontend.cpp

option(ASR_CLI_TESTS "Build complete native CLI with clearly marked host transport" OFF)
if(ASR_CLI_TESTS)
  if(NOT ASR_AUDIO_TESTS)
    message(FATAL_ERROR "ASR_CLI_TESTS requires ASR_AUDIO_TESTS=ON")
  endif()
  # No platform_identity.cc here on purpose: cli_fixture_runner.cc provides
  # the host read_native_identity double behind ASR_FIXTURE_TARGET, so the
  # real preflight in src/asr.cpp runs unchanged.
  add_executable(asr_cli_fixture ../src/main.cpp ../src/cli.cpp ../src/asr.cpp
    ../src/frontend.cpp cli_fixture_runner.cc)
  target_compile_definitions(asr_cli_fixture PRIVATE ASR_HOST_FIXTURE=1)
  target_include_directories(asr_cli_fixture PRIVATE ../inc "${ASR_SHARED_CPP}" ${ASR_JSON_INCLUDE} ${ASR_SNDFILE_INCLUDE} ${ASR_SAMPLERATE_INCLUDE})
  target_link_libraries(asr_cli_fixture PRIVATE ${ASR_SNDFILE_LIBRARY} ${ASR_SAMPLERATE_LIBRARY})
  add_test(NAME asr_cli_fixture_help COMMAND asr_cli_fixture --help)
  add_executable(test_cli_io test_cli_io.cc ../src/cli.cpp ../src/asr.cpp
    ../src/frontend.cpp "${ASR_SHARED_CPP}/platform_identity.cc")
  target_include_directories(test_cli_io PRIVATE ../inc "${ASR_SHARED_CPP}" ${ASR_JSON_INCLUDE} ${ASR_SNDFILE_INCLUDE} ${ASR_SAMPLERATE_INCLUDE})
  target_link_libraries(test_cli_io PRIVATE ${ASR_SNDFILE_LIBRARY} ${ASR_SAMPLERATE_LIBRARY})
  add_test(NAME asr_cli_io COMMAND test_cli_io "${CMAKE_CURRENT_SOURCE_DIR}/../../../test_data/vocab.json")
  foreach(target asr_cli_fixture test_cli_io)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(ASR_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endforeach()
endif()
