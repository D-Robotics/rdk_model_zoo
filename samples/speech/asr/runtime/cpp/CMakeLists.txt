cmake_minimum_required(VERSION 3.18)
project(asr_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASR_BUILD_SDK "Build native S-series SDK adapter (requires vendor SDK)" OFF)
option(ASR_BUILD_CLI "Build native ASR executable (requires ASR_BUILD_SDK)" OFF)
option(ASR_BUILD_TESTS "Build host contract tests" OFF)
find_path(ASR_SNDFILE_INCLUDE sndfile.h REQUIRED)
find_path(ASR_SAMPLERATE_INCLUDE samplerate.h REQUIRED)
find_library(ASR_SNDFILE_LIBRARY sndfile REQUIRED)
find_library(ASR_SAMPLERATE_LIBRARY samplerate REQUIRED)
get_filename_component(ASR_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../_shared/cpp" ABSOLUTE)
get_filename_component(ASR_YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vision/ultralytics_yolo/runtime/cpp" ABSOLUTE)
add_library(asr_frontend STATIC src/audio_io.cc src/frontend.cc)
target_include_directories(asr_frontend PUBLIC inc PRIVATE ${ASR_SNDFILE_INCLUDE} ${ASR_SAMPLERATE_INCLUDE})
target_link_libraries(asr_frontend PUBLIC ${ASR_SNDFILE_LIBRARY} ${ASR_SAMPLERATE_LIBRARY})
add_library(asr_preflight STATIC src/preflight.cc "${ASR_SHARED_CPP}/platform_identity.cc")
target_include_directories(asr_preflight PUBLIC inc "${ASR_SHARED_CPP}")
if(ASR_BUILD_SDK)
  find_path(ASR_DNN_INCLUDE dnn/hb_dnn.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(ASR_UCP_INCLUDE hb_ucp.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(ASR_UCP_SYS_INCLUDE hb_ucp_sys.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_library(ASR_DNN_LIBRARY dnn PATHS /usr/hobot/lib REQUIRED)
  find_library(ASR_UCP_LIBRARY hbucp PATHS /usr/hobot/lib REQUIRED)
  add_library(asr_sdk STATIC src/sdk_runner.cc "${ASR_YOLO_CPP}/common/dnn_io.cc" "${ASR_YOLO_CPP}/common/nv12_geometry.cc")
  target_include_directories(asr_sdk PUBLIC inc PRIVATE "${ASR_YOLO_CPP}" ${ASR_DNN_INCLUDE} ${ASR_UCP_INCLUDE} ${ASR_UCP_SYS_INCLUDE})
  target_link_libraries(asr_sdk PUBLIC asr_frontend asr_preflight ${ASR_DNN_LIBRARY} ${ASR_UCP_LIBRARY})
endif()
if(ASR_BUILD_CLI)
  if(NOT ASR_BUILD_SDK)
    message(FATAL_ERROR "ASR_BUILD_CLI requires ASR_BUILD_SDK=ON; host fixture is a separate test target")
  endif()
  find_path(ASR_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  add_executable(asr_demo src/main.cc src/cli_io.cc)
  target_include_directories(asr_demo PRIVATE ${ASR_JSON_INCLUDE})
  target_link_libraries(asr_demo PRIVATE asr_sdk)
endif()
if(ASR_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
