cmake_minimum_required(VERSION 3.18)
project(asr_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(ASR_BUILD_SDK "Build native S-series SDK adapter (requires vendor SDK)" OFF)
option(ASR_BUILD_CLI "Build native ASR executable (requires ASR_BUILD_SDK)" OFF)
option(ASR_BUILD_TESTS "Build host contract tests" OFF)
find_path(ASR_SNDFILE_INCLUDE sndfile.h REQUIRED)
find_path(ASR_SAMPLERATE_INCLUDE samplerate.h REQUIRED)
find_library(ASR_SNDFILE_LIBRARY sndfile REQUIRED)
find_library(ASR_SAMPLERATE_LIBRARY samplerate REQUIRED)
get_filename_component(ASR_SHARED_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../utils/c_utils" ABSOLUTE)
get_filename_component(ASR_YOLO_CPP "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vision/ultralytics_yolo/runtime/cpp" ABSOLUTE)
add_library(asr_frontend STATIC src/frontend.cpp)
target_include_directories(asr_frontend PUBLIC inc "${ASR_SHARED_CPP}" PRIVATE ${ASR_SAMPLERATE_INCLUDE})
target_link_libraries(asr_frontend PUBLIC ${ASR_SAMPLERATE_LIBRARY})
add_library(asr_model STATIC src/asr.cpp "${ASR_SHARED_CPP}/platform_identity.cc")
target_include_directories(asr_model PUBLIC inc "${ASR_SHARED_CPP}")
if(ASR_BUILD_SDK)
  find_path(ASR_DNN_INCLUDE dnn/hb_dnn.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(ASR_UCP_INCLUDE hb_ucp.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_path(ASR_UCP_SYS_INCLUDE hb_ucp_sys.h PATHS /usr/include/hobot /usr/hobot/include REQUIRED)
  find_library(ASR_DNN_LIBRARY dnn PATHS /usr/hobot/lib REQUIRED)
  find_library(ASR_UCP_LIBRARY hbucp PATHS /usr/hobot/lib REQUIRED)
  target_compile_definitions(asr_model PRIVATE ASR_ENABLE_UCP=1)
  target_sources(asr_model PRIVATE "${ASR_YOLO_CPP}/src/backend.cpp")
  target_include_directories(asr_model PRIVATE "${ASR_YOLO_CPP}" "${ASR_YOLO_CPP}/inc" ${ASR_DNN_INCLUDE} ${ASR_UCP_INCLUDE} ${ASR_UCP_SYS_INCLUDE})
  target_link_libraries(asr_model PUBLIC ${ASR_DNN_LIBRARY} ${ASR_UCP_LIBRARY})
endif()
if(ASR_BUILD_CLI)
  if(NOT ASR_BUILD_SDK)
    message(FATAL_ERROR "ASR_BUILD_CLI requires ASR_BUILD_SDK=ON; host fixture is a separate test target")
  endif()
  find_path(ASR_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  add_executable(asr_demo src/main.cpp src/cli.cpp)
  target_include_directories(asr_demo PRIVATE ${ASR_JSON_INCLUDE} ${ASR_SNDFILE_INCLUDE})
  target_link_libraries(asr_demo PRIVATE asr_model asr_frontend ${ASR_SNDFILE_LIBRARY})
endif()
if(ASR_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
