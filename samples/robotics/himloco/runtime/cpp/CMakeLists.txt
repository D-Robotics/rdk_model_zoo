cmake_minimum_required(VERSION 3.18)
project(himloco_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HIMLOCO_BUILD_SDK "Build X5 runtime against vendor libdnn" OFF)
option(HIMLOCO_BUILD_CLI "Build native CLI (requires SDK)" OFF)
option(HIMLOCO_BUILD_TESTS "Build SDK-free host policy test" OFF)
get_filename_component(HIMLOCO_SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../utils/c_utils" ABSOLUTE)
add_library(himloco_policy STATIC src/policy.cpp "${HIMLOCO_SHARED}/platform_identity.cc")
target_include_directories(himloco_policy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/inc" "${HIMLOCO_SHARED}")
if(HIMLOCO_BUILD_SDK)
  find_path(HIMLOCO_DNN_INCLUDE dnn/hb_dnn.h PATHS /usr/include /usr/include/hobot REQUIRED)
  find_library(HIMLOCO_DNN_LIBRARY dnn PATHS /usr/lib /usr/hobot/lib REQUIRED)
  target_compile_definitions(himloco_policy PRIVATE HIMLOCO_ENABLE_DNN=1)
  target_include_directories(himloco_policy PRIVATE "${HIMLOCO_DNN_INCLUDE}")
  find_package(Threads REQUIRED)
  target_link_libraries(himloco_policy PUBLIC "${HIMLOCO_DNN_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(himloco_policy PUBLIC rt)
  endif()
endif()
if(HIMLOCO_BUILD_CLI)
  if(NOT HIMLOCO_BUILD_SDK)
    message(FATAL_ERROR "HIMLOCO_BUILD_CLI requires HIMLOCO_BUILD_SDK=ON")
  endif()
  find_path(HIMLOCO_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
  add_executable(himloco_cpp src/main.cpp src/cli.cpp)
  target_include_directories(himloco_cpp PRIVATE "${HIMLOCO_JSON_INCLUDE}")
  target_link_libraries(himloco_cpp PRIVATE himloco_policy)
endif()
if(HIMLOCO_BUILD_TESTS)
  enable_testing()
  add_executable(himloco_policy_test tests/test_policy.cc)
  target_link_libraries(himloco_policy_test PRIVATE himloco_policy)
  if(MSVC)
    target_compile_options(himloco_policy_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(himloco_policy_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME himloco_policy COMMAND himloco_policy_test "${CMAKE_CURRENT_SOURCE_DIR}/../../test_data/obs_history")
endif()
